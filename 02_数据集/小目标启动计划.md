### 第2小时：编写训练代码 🏃

#### 完整训练脚本
```python
from fastai.vision.all import *
import matplotlib.pyplot as plt

# 1️⃣ 数据准备
path = Path('D:/植物识别项目/02_数据集/10植物测试集')

# 2️⃣ 数据加载
dls = ImageDataLoaders.from_folder(
    path,
    train_pct=0.8,      # 80%训练
    valid_pct=0.2,      # 20%验证
    seed=42,            # 固定随机种子
    item_tfms=Resize(224),  # 统一尺寸
    batch_tfms=augmentations  # 数据增强
)

# 3️⃣ 查看数据示例
dls.show_batch(max_n=6, figsize=(6,6))

# 4️⃣ 创建模型（使用预训练ResNet34）
learn = cnn_learner(dls, resnet34, metrics=accuracy)

# 5️⃣ 快速训练（5轮即可看到结果）
learn.fit(5)

# 6️⃣ 查看训练结果
learn.show_results()

# 7️⃣ 保存模型
learn.save('plant_model_stage1')
```

### 第3小时：测试和优化 🧪

#### 测试新图片
```python
def predict_plant(img_path):
    # 加载最佳模型
    learn.load('plant_model_stage1')
    
    # 预测
    pred, pred_idx, prob = learn.predict(img_path)
    
    print(f"预测结果：{pred}")
    print(f"置信度：{prob[pred_idx]:.2%}")
    return pred, prob[pred_idx]
```

#### 创建简单测试界面
```python
from tkinter import Tk, filedialog, Label, Button
from PIL import Image
import io

def simple_gui():
    root = Tk()
    root.title("植物识别小工具")
    
    def upload_img():
        file_path = filedialog.askopenfilename()
        if file_path:
            result = predict_plant(file_path)
            Label(root, text=f"这是：{result[0]}").pack()
    
    Button(root, text="选择图片", command=upload_img).pack()
    root.mainloop()
```
