### Python下载脚本
```python
import requests
import os
import time
from urllib.parse import quote
import json

class PlantImageDownloader:
    def __init__(self, save_path):
        self.save_path = save_path
        self.plants = ['玫瑰', '向日葵', '百合', '菊花', '兰花', 
                      '仙人掌', '竹子', '枫树', '松树', '荷花']
        
    def download_from_baidu(self, plant_name, count=30):
        """百度图片搜索下载"""
        folder = f"{self.save_path}/{plant_name}"
        os.makedirs(folder, exist_ok=True)
        
        # 搜索URL（简化版）
        search_url = f"https://image.baidu.com/search/index?tn=baiduimage&word={quote(plant_name)}"
        
        print(f"开始下载：{plant_name}")
        downloaded = 0
        
        while downloaded < count:
            try:
                # 这里需要实际解析百度图片API
                # 为演示，使用占位符URL
                img_url = f"https://example.com/{plant_name}_{downloaded}.jpg"
                
                if self.download_image(img_url, f"{folder}/{plant_name}_{downloaded}.jpg"):
                    downloaded += 1
                    
                time.sleep(0.5)  # 避免过快请求
                
            except Exception as e:
                print(f"下载出错：{e}")
                break
                
        print(f"{plant_name} 完成，下载 {downloaded} 张")
        
    def download_image(self, url, save_path):
        """下载单张图片"""
        try:
            response = requests.get(url, timeout=5)
            if response.status_code == 200:
                with open(save_path, 'wb') as f:
                    f.write(response.content)
                return True
        except:
            return False
            
    def run(self):
        """下载所有植物图片"""
        for plant in self.plants:
            self.download_from_baidu(plant, 30)
            time.sleep(2)  # 礼貌性延迟
            
# 使用方法
downloader = PlantImageDownloader("D:/植物识别项目/02_数据集/10植物测试集")
downloader.run()
```

## 🔍 数据质量检查

### 自动检查脚本
```python
from PIL import Image
import os

def check_images(path):
    """检查图片质量和数量"""
    results = {}
    
    for plant_type in os.listdir(path):
        folder = f"{path}/{plant_type}"
        if os.path.isdir(folder):
            images = []
            for img in os.listdir(folder):
                if img.endswith('.jpg') or img.endswith('.png'):
                    try:
                        img_path = f"{folder}/{img}"
                        img_file = Image.open(img_path)
                        
                        # 检查尺寸
                        width, height = img_file.size
                        if width < 224 or height < 224:
                            print(f"图片太小：{img_path}")
                            
                        # 检查文件大小
                        file_size = os.path.getsize(img_path)
                        if file_size < 10*1024:  # 小于10KB
                            print(f"文件太小：{img_path}")
                            
                        images.append(img)
                    except Exception as e:
                        print(f"损坏图片：{img_path}")
                        
            results[plant_type] = {
                'count': len(images),
                'images': images
            }
            
    return results

# 使用
results = check_images("D:/植物识别项目/02_数据集/10植物测试集")
for plant, info in results.items():
    print(f"{plant}: {info['count']} 张图片")
```

## 🎨 图片预处理

### 自动调整脚本
```python
from PIL import Image
import glob
import os

def preprocess_images(source_path, target_path):
    """统一处理所有图片"""
    os.makedirs(target_path, exist_ok=True)
    
    for plant_type in os.listdir(source_path):
        plant_folder = f"{source_path}/{plant_type}"
        target_folder = f"{target_path}/{plant_type}"
        os.makedirs(target_folder, exist_ok=True)
        
        images = glob.glob(f"{plant_folder}/*.jpg")
        
        for img_path in images:
            try:
                # 打开图片
                img = Image.open(img_path)
                
                # 调整尺寸为224x224
                img = img.resize((224, 224), Image.Resampling.LANCZOS)
                
                # 保存
                filename = os.path.basename(img_path)
                img.save(f"{target_folder}/{filename}")
                
                print(f"处理完成：{filename}")
            except Exception as e:
                print(f"处理失败：{img_path}, 错误：{e}")
```
